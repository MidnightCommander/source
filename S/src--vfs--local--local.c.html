  63 static void *
  64 local_open (const vfs_path_t *vpath, int flags, mode_t mode)
     /*   */
  65 {
  66     int *local_info;
  67     int fd;
  68     const char *path;
  69 
  70     path = vfs_path_get_last_path_str (vpath);
  71     fd = open (path, NO_LINEAR (flags), mode);
  72     if (fd == -1)
  73         return NULL;
  74 
  75     local_info = g_new (int, 1);
  76     *local_info = fd;
  77 
  78     return local_info;

  83 static void *
  84 local_opendir (const vfs_path_t *vpath)
     /*   */
  85 {
  86     DIR *dir = NULL;
  87 
  88     const char *path = vfs_path_get_last_path_str (vpath);
  89 
  90     /* According to POSIX, `opendir` and `readdir` can't return EINTR (unlike `closedir`).
  91      *
  92      * Unfortunately, in practice, at least `readdir` is known to return EINTR on Linux >= 5.1 on
  93      * networked filesystems such as CIFS. Networked FUSE-based systems appear to be affected as
  94      * well. What's worse is that when you retry `readdir`, the file list still comes out empty.
  95      *
  96      * So our only option seems to be to try `readdir` immediately after `opendir` and reopen the
  97      * directory if it failed with EINTR, which usually leads to correct file listing, or use
  98      * `rewinddir` otherwise, even if `readdir` failed, since FreeBSD and macOS may have advanced
  99      * the stream anyway.
 100      *
 101      * However, this has caused problems on FUSE-based systems that do not properly implement
 102      * `rewinddir` in the past. No silver bullet...
 103      */
 104     while (dir == NULL)
 105     {
 106         dir = opendir (path);
 107         if (dir == NULL)
 108         {
 109             // Unlike readdir(), opendir() always sets errno on failure, so no reset is needed
 110             if (errno == EINTR)
 111                 continue;
 112 
 113             return NULL;
 114         }
 115 
 116         // Reset errno before readdir to avoid reading a stale EINTR (#5156)
 117         errno = 0;
 118 
 119         if (readdir (dir) == NULL && errno == EINTR)
 120         {
 121             closedir (dir);
 122             dir = NULL;
 123         }
 124         else
 125             rewinddir (dir);
 126     }
 127 
 128     DIR **local_info = (DIR **) g_new (DIR *, 1);
 129     *local_info = dir;
 130 
 131     return local_info;
 132 }

 136 static struct vfs_dirent *
 137 local_readdir (void *data)
     /*   */
 138 {
 139     struct dirent *d;
 140     unsigned char type;
 141 
 142     // Retry if interrupted, resetting errno to avoid reading a stale EINTR (#5156). FreeBSD and
 143     // macOS may lose the entries read before the interruption, but rewinding would repeat the
 144     // ones already returned.
 145     do
 146     {
 147         errno = 0;
 148         d = readdir (*(DIR **) data);
 149     }
 150     while (d == NULL && errno == EINTR);
 151 
 152     if (d == NULL)
 153         return NULL;
 154 
 155 #ifdef HAVE_STRUCT_DIRENT_D_TYPE
 156     type = d->d_type;
 157 #else
 158     type = DT_UNKNOWN;
 159 #endif
 160 
 161     return vfs_dirent_init (NULL, d->d_name, d->d_ino, type);
 162 }

 294 static ssize_t
 295 local_write (void *data, const char *buf, size_t nbyte)
     /*   */
 296 {
 297     int fd;
 298     int n;
 299 
 300     if (data == NULL)
 301         return (-1);
 302 
 303     fd = *(int *) data;
 304 
 305     while ((n = write (fd, buf, nbyte)) == -1)
 306     {
 307 #ifdef EAGAIN
 308         if (errno == EAGAIN)
 309             continue;
 310 #endif
 311 #ifdef EINTR
 312         if (errno == EINTR)
 313             continue;
 314 #endif
 315         break;
 316     }
 317 
 318     return n;
 319 }

 424 ssize_t
 425 local_read (void *data, char *buffer, size_t count)
     /*   */
 426 {
 427     int n;
 428     int fd;
 429 
 430     if (data == NULL)
 431         return (-1);
 432 
 433     fd = *(int *) data;
 434 
 435     while ((n = read (fd, buffer, count)) == -1)
 436     {
 437 #ifdef EAGAIN
 438         if (errno == EAGAIN)
 439             continue;
 440 #endif
 441 #ifdef EINTR
 442         if (errno == EINTR)
 443             continue;
 444 #endif
 445         return (-1);
 446     }
 447 
 448     return n;
 449 }

 118 static tree_entry *
 119 back_ptr (tree_entry *ptr, int *count)
     /*   */
 120 {
 121     int i;
 122 
 123     for (i = 0; ptr != NULL && ptr->prev != NULL && i < *count; ptr = ptr->prev, i++)
 124         ;
 125 
 126     *count = i;
 127     return ptr;
 128 }

 265 static void
 266 show_tree (WTree *tree)
     /*   */
 267 {
 268     Widget *w = WIDGET (tree);
 269     tree_entry *current;
 270     int i, j;
 271     int topsublevel = 0;
 272     int x = 0, y = 0;
 273     int tree_lines, tree_cols;
 274 
 275     // Initialize
 276     tree_lines = tlines (tree);
 277     tree_cols = w->rect.cols;
 278 
 279     widget_gotoyx (w, y, x);
 280     if (tree->is_panel)
 281     {
 282         tree_cols -= 2;
 283         x = y = 1;
 284     }
 285 
 286     g_free (tree->tree_shown);
 287     tree->tree_shown = g_new0 (tree_entry *, tree_lines);
 288 
 289     if (tree->store->tree_first != NULL)
 290         topsublevel = tree->store->tree_first->sublevel;
 291 
 292     if (tree->selected_ptr == NULL)
 293     {
 294         tree->selected_ptr = tree->store->tree_first;
 295         tree->topdiff = 0;
 296     }
 297     current = tree->selected_ptr;
 298 
 299     // Calculate the directory which is to be shown on the topmost line
 300     if (!tree_navigation_flag)
 301         current = back_ptr (current, &tree->topdiff);
 302     else
 303     {
 304         i = 0;
 305 
 306         while (current->prev != NULL && i < tree->topdiff)
 307         {
 308             current = current->prev;
 309 
 310             if (current->sublevel < tree->selected_ptr->sublevel)
 311             {
 312                 if (vfs_path_equal (current->name, tree->selected_ptr->name))
 313                     i++;
 314             }
 315             else if (current->sublevel == tree->selected_ptr->sublevel)
 316             {
 317                 const char *cname;
 318 
 319                 cname = vfs_path_as_str (current->name);
 320                 for (j = strlen (cname) - 1; !IS_PATH_SEP (cname[j]); j--)
 321                     ;
 322                 if (vfs_path_equal_len (current->name, tree->selected_ptr->name, j))
 323                     i++;
 324             }
 325             else if (current->sublevel == tree->selected_ptr->sublevel + 1)
 326             {
 327                 j = vfs_path_len (tree->selected_ptr->name);
 328                 if (j > 1 && vfs_path_equal_len (current->name, tree->selected_ptr->name, j))
 329                     i++;
 330             }
 331         }
 332         tree->topdiff = i;
 333     }
 334 
 335     // Loop for every line
 336     for (i = 0; i < tree_lines; i++)
 337     {
 338         const int *colors;
 339 
 340         colors = widget_get_colors (w);
 341         tty_setcolor (tree->is_panel ? CORE_NORMAL_COLOR : colors[DLG_COLOR_NORMAL]);
 342 
 343         // Move to the beginning of the line
 344         tty_draw_hline (w->rect.y + y + i, w->rect.x + x, ' ', tree_cols);
 345 
 346         if (current == NULL)
 347             continue;
 348 
 349         if (tree->is_panel)
 350         {
 351             gboolean selected;
 352 
 353             selected = widget_get_state (w, WST_FOCUSED) && current == tree->selected_ptr;
 354             tty_setcolor (selected ? CORE_SELECTED_COLOR : CORE_NORMAL_COLOR);
 355         }
 356         else
 357         {
 358             int idx = current == tree->selected_ptr ? DLG_COLOR_FOCUS : DLG_COLOR_NORMAL;
 359 
 360             tty_setcolor (colors[idx]);
 361         }
 362 
 363         tree->tree_shown[i] = current;
 364         if (current->sublevel == topsublevel)
 365             // Show full name
 366             tty_print_string (str_fit_to_term (vfs_path_as_str (current->name),
 367                                                tree_cols + (tree->is_panel ? 0 : 1), J_LEFT_FIT));
 368         else
 369         {
 370             // Output branch parts
 371             for (j = 0; j < current->sublevel - topsublevel - 1; j++)
 372             {
 373                 if (tree_cols - 8 - 3 * j < 9)
 374                     break;
 375                 tty_print_char (' ');
 376                 if ((current->submask & (1 << (j + topsublevel + 1))) != 0)
 377                     tty_print_char (mc_tty_frm[MC_TTY_FRM_VERT]);
 378                 else
 379                     tty_print_char (' ');
 380                 tty_print_char (' ');
 381             }
 382             tty_print_char (' ');
 383             j++;
 384             if (current->next == NULL || (current->next->submask & (1 << current->sublevel)) == 0)
 385                 tty_print_char (mc_tty_frm[MC_TTY_FRM_LEFTBOTTOM]);
 386             else
 387                 tty_print_char (mc_tty_frm[MC_TTY_FRM_LEFTMIDDLE]);
 388             tty_print_char (mc_tty_frm[MC_TTY_FRM_HORIZ]);
 389 
 390             // Show sub-name
 391             tty_print_char (' ');
 392             tty_print_string (
 393                 str_fit_to_term (current->subname, tree_cols - x - 3 * j, J_LEFT_FIT));
 394         }
 395 
 396         // Calculate the next value for current
 397         current = current->next;
 398         if (tree_navigation_flag)
 399             for (; current != NULL; current = current->next)
 400             {
 401                 if (current->sublevel < tree->selected_ptr->sublevel)
 402                 {
 403                     if (vfs_path_equal_len (current->name, tree->selected_ptr->name,
 404                                             vfs_path_len (current->name)))
 405                         break;
 406                 }
 407                 else if (current->sublevel == tree->selected_ptr->sublevel)
 408                 {
 409                     const char *cname;
 410 
 411                     cname = vfs_path_as_str (current->name);
 412                     for (j = strlen (cname) - 1; !IS_PATH_SEP (cname[j]); j--)
 413                         ;
 414                     if (vfs_path_equal_len (current->name, tree->selected_ptr->name, j))
 415                         break;
 416                 }
 417                 else if (current->sublevel == tree->selected_ptr->sublevel + 1
 418                          && vfs_path_len (tree->selected_ptr->name) > 1)
 419                 {
 420                     if (vfs_path_equal_len (current->name, tree->selected_ptr->name,
 421                                             vfs_path_len (tree->selected_ptr->name)))
 422                         break;
 423                 }
 424             }
 425     }
 426 
 427     tree_show_mini_info (tree, tree_lines, tree_cols);
 428 }

 443 static void
 444 tree_move_backward (WTree *tree, int i)
     /*   */
 445 {
 446     if (!tree_navigation_flag)
 447         tree->selected_ptr = back_ptr (tree->selected_ptr, &i);
 448     else
 449     {
 450         tree_entry *current;
 451         int j = 0;
 452 
 453         current = tree->selected_ptr;
 454         while (j < i && current->prev != NULL
 455                && current->prev->sublevel >= tree->selected_ptr->sublevel)
 456         {
 457             current = current->prev;
 458             if (current->sublevel == tree->selected_ptr->sublevel)
 459             {
 460                 tree->selected_ptr = current;
 461                 j++;
 462             }
 463         }
 464         i = j;
 465     }
 466 
 467     tree->topdiff -= i;
 468     tree_check_focus (tree);
 469 }

     /*   */
 475 {
 476     if (!tree_navigation_flag)
 477         tree->selected_ptr = forw_ptr (tree->selected_ptr, &i);
 478     else
 479     {
 480         tree_entry *current;
 481         int j = 0;
 482 
 483         current = tree->selected_ptr;
 484         while (j < i && current->next != NULL
 485                && current->next->sublevel >= tree->selected_ptr->sublevel)
 486         {
 487             current = current->next;
 488             if (current->sublevel == tree->selected_ptr->sublevel)
 489             {
 490                 tree->selected_ptr = current;
 491                 j++;
 492             }
 493         }
 494         i = j;
 495     }
 496 
 497     tree->topdiff += i;
 498     tree_check_focus (tree);

 503 static void
 504 tree_move_to_child (WTree *tree)
     /*   */
 505 {
 506     tree_entry *current;
 507 
 508     // Do we have a starting point?
 509     if (tree->selected_ptr == NULL)
 510         return;
 511 
 512     // Take the next entry
 513     current = tree->selected_ptr->next;
 514     // Is it the child of the selected entry
 515     if (current != NULL && current->sublevel > tree->selected_ptr->sublevel)
 516     {
 517         // Yes -> select this entry
 518         tree->selected_ptr = current;
 519         tree->topdiff++;
 520         tree_check_focus (tree);
 521     }
 522     else
 523     {
 524         // No -> rescan and try again
 525         tree_rescan (tree);
 526         current = tree->selected_ptr->next;
 527         if (current != NULL && current->sublevel > tree->selected_ptr->sublevel)
 528         {
 529             tree->selected_ptr = current;
 530             tree->topdiff++;
 531             tree_check_focus (tree);
 532         }
 533     }
 534 }

 538 static gboolean
 539 tree_move_to_parent (WTree *tree)
     /*   */
 540 {
 541     tree_entry *current;
 542     tree_entry *old;
 543 
 544     if (tree->selected_ptr == NULL)
 545         return FALSE;
 546 
 547     old = tree->selected_ptr;
 548 
 549     for (current = tree->selected_ptr->prev;
 550          current != NULL && current->sublevel >= tree->selected_ptr->sublevel;
 551          current = current->prev)
 552         tree->topdiff--;
 553 
 554     if (current == NULL)
 555         current = tree->store->tree_first;
 556     tree->selected_ptr = current;
 557     tree_check_focus (tree);
 558     return tree->selected_ptr != old;
 559 }

 620 static gboolean
 621 search_tree (WTree *tree, const GString *text)
     /*   */
 622 {
 623     tree_entry *current = tree->selected_ptr;
 624     gboolean wrapped = FALSE;
 625     gboolean found = FALSE;
 626 
 627     while (!found && (!wrapped || current != tree->selected_ptr))
 628         if (strncmp (current->subname, text->str, text->len) == 0)
 629         {
 630             tree->selected_ptr = current;
 631             found = TRUE;
 632         }
 633         else
 634         {
 635             current = current->next;
 636             if (current == NULL)
 637             {
 638                 current = tree->store->tree_first;
 639                 wrapped = TRUE;
 640             }
 641 
 642             tree->topdiff++;
 643         }
 644 
 645     tree_check_focus (tree);
 646     return found;
 647 }

     /*   */
 653 {

 658     else if (key != 0)
 659         g_string_append_c (tree->search_buffer, (gchar) key);

 662         g_string_set_size (tree->search_buffer, tree->search_buffer->len - 1);

 664     show_tree (tree);
 665     maybe_chdir (tree);

 670 static void
 671 tree_rescan (void *data)
     /*   */
 672 {
 673     WTree *tree = data;
 674     vfs_path_t *old_vpath;
 675 
 676     old_vpath = vfs_path_clone (vfs_get_raw_current_dir ());
 677     if (old_vpath == NULL)
 678         return;
 679 
 680     if (tree->selected_ptr != NULL && mc_chdir (tree->selected_ptr->name) == 0)
 681     {
 682         int ret;
 683 
 684         tree_store_rescan (tree->selected_ptr->name);
 685         ret = mc_chdir (old_vpath);
 686         (void) ret;
 687     }
 688     vfs_path_free (old_vpath, TRUE);
 689 }

 704 static void
 705 tree_copy (WTree *tree, const char *default_dest)
     /*   */
 706 {
 707     char msg[BUF_MEDIUM];
 708     char *dest;
 709 
 710     if (tree->selected_ptr == NULL)
 711         return;
 712

 755

 757             file_error_message (_ ("Cannot stat the destination\n%s"), dest);

 768             file_progress_ui_create (ctx, FALSE, FILEGUI_DIALOG_ONE_ITEM);

 770             file_op_context_destroy (ctx);
 771         }
 772 
 773         vfs_path_free (dest_vpath, TRUE);
 774     }
 775 
 776     g_free (dest);

     /*   */
 927 {

 931             tree_move_to_top (tree);

 950         tree->searching = TRUE;

 984 static cb_ret_t
 985 tree_execute_cmd (WTree *tree, long command)
     /*   */
 986 {
 987     cb_ret_t res = MSG_HANDLED;
 988 
 989     if (command != CK_Search)
 990         tree->searching = FALSE;
 991 
 992     switch (command)
 993     {
 994     case CK_Help:
 995         tree_help ();
 996         break;
 997     case CK_Forget:
 998         tree_forget (tree);
 999         break;
1000     case CK_ToggleNavigation:
1001         tree_toggle_navig (tree);
1002         break;
1003     case CK_Copy:
1004         tree_copy (tree, "");
1005         break;
1006     case CK_Move:
1007         tree_move (tree, "");
1008         break;
1009     case CK_Up:
1010         tree_move_up (tree);
1011         break;
1012     case CK_Down:
1013         tree_move_down (tree);
1014         break;
1015     case CK_Top:
1016         tree_move_home (tree);
1017         break;
1018     case CK_Bottom:
1019         tree_move_end (tree);
1020         break;
1021     case CK_PageUp:
1022         tree_move_pgup (tree);
1023         break;
1024     case CK_PageDown:
1025         tree_move_pgdn (tree);
1026         break;
1027     case CK_Enter:
1028         tree_chdir_sel (tree);
1029         break;
1030     case CK_Reread:
1031         tree_rescan (tree);
1032         break;
1033     case CK_Search:
1034         tree_start_search (tree);
1035         break;
1036     case CK_Delete:
1037         tree_rmdir (tree);
1038         break;
1039     case CK_Quit:
1040         if (!tree->is_panel)
1041             dlg_close (DIALOG (WIDGET (tree)->owner));
1042         return res;
1043     default:
1044         res = MSG_NOT_HANDLED;
1045     }
1046 
1047     show_tree (tree);
1048 
1049     return res;
1050 }

1053 
1054 static cb_ret_t
1055 tree_key (WTree *tree, int key)
     /*   */
1056 {
1057     long command;
1058 
1059     if (is_abort_char (key))
1060     {
1061         if (tree->is_panel)
1062         {
1063             tree->searching = FALSE;
1064             show_tree (tree);
1065             return MSG_HANDLED;  // eat abort char
1066         }
1067         /* modal tree dialog: let upper layer see the
1068            abort character and close the dialog */
1069         return MSG_NOT_HANDLED;
1070     }
1071 
1072     if (tree->searching && ((key >= ' ' && key <= 255) || key == KEY_BACKSPACE))
1073     {
1074         tree_do_search (tree, key);
1075         show_tree (tree);
1076         return MSG_HANDLED;
1077     }
1078 
1079     command = widget_lookup_key (WIDGET (tree), key);
1080     switch (command)
1081     {
1082     case CK_IgnoreKey:
1083         break;
1084     case CK_Left:
1085         return tree_move_left (tree) ? MSG_HANDLED : MSG_NOT_HANDLED;
1086     case CK_Right:
1087         return tree_move_right (tree) ? MSG_HANDLED : MSG_NOT_HANDLED;
1088     default:
1089         tree_execute_cmd (tree, command);
1090         return MSG_HANDLED;
1091     }
1092 
1093     // Do not eat characters not meant for the tree below ' ' (e.g. C-l).
1094     if (!mc_global.cmdline_visible && ((key >= ' ' && key <= 255) || key == KEY_BACKSPACE))
1095     {
1096         tree_start_search (tree);
1097         tree_do_search (tree, key);
1098         return MSG_HANDLED;
1099     }
1100 
1101     return MSG_NOT_HANDLED;
1102 }

     /*   */

1214     int y;
1215

1218         y--;
1219

1223         // rest of the upper frame - call menu
1224         if (tree->is_panel && event->y == WIDGET (w->owner)->rect.y)
1225         {
1226             // return MOU_UNHANDLED
1227             event->result.abort = TRUE;
1228         }
1229         else if (!widget_get_state (w, WST_FOCUSED))
1230             (void) change_panel ();
1231         break;
1232 
1233     case MSG_MOUSE_CLICK:
1234     {
1235         int lines;
1236 
1237         lines = tlines (tree);
1238 
1239         if (y < 0)
1240         {
1241             tree_move_backward (tree, lines - 1);
1242             show_tree (tree);
1243         }
1244         else if (y >= lines)
1245         {
1246             tree_move_forward (tree, lines - 1);
1247             show_tree (tree);
1248         }
1249         else if ((event->count & GPM_DOUBLE) != 0)
1250         {
1251             if (tree->tree_shown[y] != NULL)
1252             {
1253                 tree->selected_ptr = tree->tree_shown[y];
1254                 tree->topdiff = y;
1255             }
1256

1258         }
1259     }
1260     break;
1261 
1262     case MSG_MOUSE_SCROLL_UP:
1263     case MSG_MOUSE_SCROLL_DOWN:
1264         // TODO: Ticket #2218

1266

1269     }

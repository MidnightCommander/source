 117 /**
 118  * Run viewer (internal or external) on the current file.
 119  * If @plain_view is TRUE, force internal viewer and raw mode (used for F13).
 120  */
 121 static void
 122 do_view_cmd (WPanel *panel, gboolean plain_view)
     /*   */
 123 {
 124     const file_entry_t *fe;
 125 
 126     fe = panel_current_entry (panel);
 127     if (fe == NULL)
 128         return;
 129 
 130     // Directories are viewed by changing to them
 131     if (S_ISDIR (fe->st.st_mode) || link_isdir (fe))
 132     {
 133         vfs_path_t *fname_vpath;
 134 
 135         if (confirm_view_dir && (panel->marked != 0 || panel->dirs_marked != 0)
 136             && query_dialog (_ ("Confirmation"), _ ("Files tagged, want to cd?"), D_NORMAL, 2,
 137                              _ ("&Yes"), _ ("&No"))
 138                 != 0)
 139             return;
 140 
 141         fname_vpath = vfs_path_from_str (fe->fname->str);
 142         if (!panel_cd (panel, fname_vpath, cd_exact))
 143             cd_error_message (fe->fname->str);
 144         vfs_path_free (fname_vpath, TRUE);
 145     }
 146     else
 147     {
 148         vfs_path_t *filename_vpath;
 149 
 150         filename_vpath = vfs_path_from_str (fe->fname->str);
 151         view_file (filename_vpath, plain_view, use_internal_view);
 152         vfs_path_free (filename_vpath, TRUE);
 153     }
 154 
 155     repaint_screen ();
 156 }

 168 /**
 169  * Read from @fd until @buf is full or the end of the file is reached: read() may return less
 170  * than requested before the end of the file (network and FUSE/FSKit file systems, VFS).
 171  *
 172  * @return number of bytes read, -1 on error
 173  */
 174 
 175 static ssize_t
 176 compare_read (int fd, char *buf, size_t size)
     /*   */
 177 {
 178     size_t done = 0;
 179 
 180     while (done < size)
 181     {
 182         const ssize_t n = mc_read (fd, buf + done, size - done);
 183 
 184         if (n < 0)
 185         {
 186             if (errno == EINTR)
 187                 continue;
 188             return -1;
 189         }
 190         if (n == 0)
 191             break;
 192         done += (size_t) n;
 193     }
 194 
 195     return (ssize_t) done;
 196 }

 199 /**
 200  * Compare the contents of two files, which can be on any VFS.
 201  *
 202  * @return 0 if the files are identical, non-zero if they differ or can't be read
 203  */
 204 
 205 static int
 206 compare_files (const vfs_path_t *vpath1, const vfs_path_t *vpath2, off_t size)
     /*   */
 207 {
 208     int file1;
 209     int result = -1;  // Different by default
 210 
 211     if (size == 0)
 212         return 0;
 213 
 214     file1 = mc_open (vpath1, O_RDONLY);
 215     if (file1 >= 0)
 216     {
 217         int file2;
 218 
 219         file2 = mc_open (vpath2, O_RDONLY);
 220         if (file2 >= 0)
 221         {
 222             char *buf1, *buf2;
 223             ssize_t n1, n2;
 224 
 225             buf1 = g_malloc (IO_BUFSIZE);
 226             buf2 = g_malloc (IO_BUFSIZE);
 227 
 228             rotate_dash (TRUE);
 229             do
 230             {
 231                 n1 = compare_read (file1, buf1, IO_BUFSIZE);
 232                 n2 = compare_read (file2, buf2, IO_BUFSIZE);
 233             }
 234             while (n1 == IO_BUFSIZE && n2 == IO_BUFSIZE && memcmp (buf1, buf2, IO_BUFSIZE) == 0);
 235             result = n1 < 0 || n2 < 0 || n1 != n2 || memcmp (buf1, buf2, (size_t) n1) != 0;
 236             rotate_dash (FALSE);
 237 
 238             g_free (buf1);
 239             g_free (buf2);
 240             mc_close (file2);
 241         }
 242         mc_close (file1);
 243     }
 244 
 245     return result;
 246 }

 250 static void
 251 compare_dir (WPanel *panel, const WPanel *other, enum CompareMode mode)
     /*   */
 252 {
 253     int i, j;
 254 
 255     // No marks by default
 256     panel->marked = 0;
 257     panel->total = 0;
 258     panel->dirs_marked = 0;
 259 
 260     // Handle all files in the panel
 261     for (i = 0; i < panel->dir.len; i++)
 262     {
 263         file_entry_t *source = &panel->dir.list[i];
 264         const char *source_fname;
 265 
 266         // Default: unmarked
 267         file_mark (panel, i, 0);
 268 
 269         // Skip directories
 270         if (S_ISDIR (source->st.st_mode))
 271             continue;
 272 
 273         source_fname = source->fname->str;
 274         if (panel->is_panelized)
 275             source_fname = x_basename (source_fname);
 276 
 277         // Search the corresponding entry from the other panel
 278         for (j = 0; j < other->dir.len; j++)
 279         {
 280             const char *other_fname;
 281 
 282             other_fname = other->dir.list[j].fname->str;
 283             if (other->is_panelized)
 284                 other_fname = x_basename (other_fname);
 285 
 286             if (strcmp (source_fname, other_fname) == 0)
 287                 break;
 288         }
 289 
 290         if (j >= other->dir.len)
 291             // Not found -> mark
 292             do_file_mark (panel, i, 1);
 293         else
 294         {
 295             // Found
 296             file_entry_t *target = &other->dir.list[j];
 297 
 298             if (mode != compare_size_only)
 299                 // Older version is not marked
 300                 if (source->st.st_mtime < target->st.st_mtime)
 301                     continue;
 302 
 303             // Newer version with different size is marked
 304             if (source->st.st_size != target->st.st_size)
 305             {
 306                 do_file_mark (panel, i, 1);
 307                 continue;
 308             }
 309 
 310             if (mode == compare_size_only)
 311                 continue;
 312 
 313             if (mode == compare_quick)
 314             {
 315                 // Thorough compare off, compare only time stamps
 316                 // Mark newer version, don't mark version with the same date
 317                 if (source->st.st_mtime > target->st.st_mtime)
 318                     do_file_mark (panel, i, 1);
 319 
 320                 continue;
 321             }
 322 
 323             // Thorough compare on, do byte-by-byte comparison
 324             {
 325                 vfs_path_t *src_name, *dst_name;
 326 
 327                 src_name =
 328                     vfs_path_append_new (panel->cwd_vpath, source->fname->str, (char *) NULL);
 329                 dst_name =
 330                     vfs_path_append_new (other->cwd_vpath, target->fname->str, (char *) NULL);
 331                 if (compare_files (src_name, dst_name, source->st.st_size))
 332                     do_file_mark (panel, i, 1);
 333                 vfs_path_free (src_name, TRUE);
 334                 vfs_path_free (dst_name, TRUE);
 335             }
 336         }
 337     }  // for (i ...)
 338 }

 342 static void
 343 do_link (link_type_t link_type, const char *fname)
     /*   */
 344 {
 345     char *dest = NULL, *src = NULL;
 346     vfs_path_t *dest_vpath = NULL;
 347 
 348     if (link_type == LINK_HARDLINK)
 349     {
 350         vfs_path_t *fname_vpath;
 351 
 352         src = g_strdup_printf (_ ("Link %s to:"), str_trunc (fname, 46));
 353         dest =
 354             input_expand_dialog (_ ("Link"), src, MC_HISTORY_FM_LINK, "", INPUT_COMPLETE_FILENAMES);
 355         if (dest == NULL || *dest == '\0')
 356             goto cleanup;
 357 
 358         save_cwds_stat ();
 359 
 360         fname_vpath = vfs_path_from_str (fname);
 361         dest_vpath = vfs_path_from_str (dest);
 362         if (mc_link (fname_vpath, dest_vpath) == -1)
 363             file_error_message (_ ("Cannot create link\n%s"), dest);
 364         vfs_path_free (fname_vpath, TRUE);
 365     }
 366     else
 367     {
 368         vfs_path_t *s, *d;
 369 
 370         /* suggest the full path for symlink, and either the full or
 371            relative path to the file it points to  */
 372         s = vfs_path_append_new (current_panel->cwd_vpath, fname, (char *) NULL);
 373 
 374         if (get_other_type () == view_listing)
 375             d = vfs_path_append_new (other_panel->cwd_vpath, fname, (char *) NULL);
 376         else
 377             d = vfs_path_from_str (fname);
 378 
 379         if (link_type == LINK_SYMLINK_RELATIVE)
 380         {
 381             char *s_str;
 382 
 383             s_str = diff_two_paths (other_panel->cwd_vpath, s);
 384             vfs_path_free (s, TRUE);
 385             s = vfs_path_from_str_flags (s_str, VPF_NO_CANON);
 386             g_free (s_str);
 387         }
 388 
 389         symlink_box (s, d, &dest, &src);
 390         vfs_path_free (d, TRUE);
 391         vfs_path_free (s, TRUE);
 392 
 393         if (dest == NULL || *dest == '\0' || src == NULL || *src == '\0')
 394             goto cleanup;
 395 
 396         save_cwds_stat ();
 397 
 398         dest_vpath = vfs_path_from_str_flags (dest, VPF_NO_CANON);
 399 
 400         s = vfs_path_from_str (src);
 401         if (mc_symlink (dest_vpath, s) == -1)
 402             file_error_message (_ ("Cannot create symbolic link\n%s"), dest);
 403         vfs_path_free (s, TRUE);
 404     }
 405 
 406     update_panels (UP_OPTIMIZE, UP_KEEPSEL);
 407     repaint_screen ();
 408 
 409 cleanup:
 410     vfs_path_free (dest_vpath, TRUE);
 411     g_free (src);
 412     g_free (dest);
 413 }

     /*   */
 420          const char *prefix, int to_home, gboolean strip_password)

 459         if (!panel_do_cd (MENU_PANEL, cd_vpath, cd_parse_command))

 462

 464                 create_panel (MENU_PANEL_IDX, save_type);
 465         }
 466         vfs_path_free (cd_vpath, TRUE);
 467     }
 468     g_free (cd_path);
 469 
 470     // In case of passive panel, restore current VFS directory that was changed in panel_do_cd()
 471     if (MENU_PANEL != current_panel)
 472         (void) mc_chdir (current_panel->cwd_vpath);

 518 gboolean
 519 view_file_at_line (const vfs_path_t *filename_vpath, gboolean plain_view, gboolean internal,
     /*   */
 520                    long start_line, off_t search_start, off_t search_end)
 521 {
 522     gboolean ret = TRUE;
 523 
 524     if (plain_view)
 525     {
 526         mcview_mode_flags_t changed_flags;
 527 
 528         mcview_clear_mode_flags (&changed_flags);
 529         mcview_altered_flags.hex = FALSE;
 530         mcview_altered_flags.magic = FALSE;
 531         mcview_altered_flags.nroff = FALSE;
 532         if (mcview_global_flags.hex)
 533             changed_flags.hex = TRUE;
 534         if (mcview_global_flags.magic)
 535             changed_flags.magic = TRUE;
 536         if (mcview_global_flags.nroff)
 537             changed_flags.nroff = TRUE;
 538         mcview_global_flags.hex = FALSE;
 539         mcview_global_flags.magic = FALSE;
 540         mcview_global_flags.nroff = FALSE;
 541 
 542         ret = mcview_viewer (NULL, filename_vpath, start_line, search_start, search_end);
 543 
 544         if (changed_flags.hex && !mcview_altered_flags.hex)
 545             mcview_global_flags.hex = TRUE;
 546         if (changed_flags.magic && !mcview_altered_flags.magic)
 547             mcview_global_flags.magic = TRUE;
 548         if (changed_flags.nroff && !mcview_altered_flags.nroff)
 549             mcview_global_flags.nroff = TRUE;
 550 
 551         dialog_switch_process_pending ();
 552     }
 553     else if (internal)
 554     {
 555         char view_entry[BUF_TINY];
 556 
 557         if (start_line > 0)
 558             g_snprintf (view_entry, sizeof (view_entry), "View:%ld", start_line);
 559         else
 560             strcpy (view_entry, "View");
 561 
 562         ret = (regex_command (filename_vpath, view_entry) == 0);
 563         if (ret)
 564         {
 565             ret = mcview_viewer (NULL, filename_vpath, start_line, search_start, search_end);
 566             dialog_switch_process_pending ();
 567         }
 568     }
 569     else
 570     {
 571         static const char *viewer = NULL;
 572 
 573         if (viewer == NULL)
 574         {
 575             viewer = getenv ("VIEWER");
 576             if (viewer == NULL)
 577                 viewer = getenv ("PAGER");
 578             if (viewer == NULL)
 579                 viewer = "view";
 580         }
 581 
 582         execute_external_editor_or_viewer (viewer, filename_vpath, start_line);
 583     }
 584 
 585     return ret;
 586 }

 649 view_filtered_cmd (const WPanel *panel)
     /*   */
 650 {
 651     char *command;
 652     const char *initial_command;
 653 
 654     if (input_is_empty (the_cmdline))
 655     {

 657

 797     {
 798         vfs_path_t *absdir;

 801             absdir = vfs_path_from_str (dir);

 803         {
 804             // possible escaped '~'
 805             // allow create directory with name '~'
 806             char *tmpdir = dir;
 807

 809                 tmpdir = dir + 1;

 812         }
 813 
 814         save_cwds_stat ();
 815 
 816         if (mc_mkdir (absdir, 0777) != 0)
 817             file_error_message (_ ("Cannot create directory\n%s"), vfs_path_as_str (absdir));
 818         else
 819         {
 820             update_panels (UP_OPTIMIZE, dir);
 821             repaint_screen ();
 822             select_item (panel);
 823         }
 824 
 825         vfs_path_free (absdir, TRUE);

1158                                             vfs_path_as_str (dest_vpath));

     /*   */
1263 {
1264     file_entry_t *entry;
1265 
1266     entry = panel_current_entry (panel);
1267 
1268     if (entry != NULL && S_ISDIR (entry->st.st_mode) && !DIR_IS_DOTDOT (entry->fname->str))
1269     {
1270         size_t dir_count = 0;
1271         size_t count = 0;
1272         uintmax_t total = 0;
1273         dirsize_status_msg_t dsm;
1274         vfs_path_t *p;
1275 
1276         p = vfs_path_from_str (entry->fname->str);
1277 
1278         memset (&dsm, 0, sizeof (dsm));
1279         status_msg_init (STATUS_MSG (&dsm), _ ("Directory scanning"), 0, dirsize_status_init_cb,
1280                          dirsize_status_update_cb, dirsize_status_deinit_cb);
1281 
1282         if (compute_dir_size (p, &dsm, &dir_count, &count, &total, FALSE) == FILE_CONT)
1283         {
1284             entry->st.st_size = (off_t) total;
1285             entry->f.dir_size_computed = 1;
1286         }
1287 
1288         vfs_path_free (p, TRUE);
1289 
1290         status_msg_deinit (STATUS_MSG (&dsm));
1291     }
1292 
1293     if (panels_options.mark_moves_down)
1294         send_message (panel, NULL, MSG_ACTION, CK_Down, NULL);
1295 
1296     recalculate_panel_summary (panel);
1297 
1298     if (panel->sort_field->sort_routine == (GCompareFunc) sort_size)
1299         panel_re_sort (panel);
1300 
1301     panel->dirty = TRUE;

1306 void
1307 dirsizes_cmd (WPanel *panel)
     /*   */
1308 {
1309     int i;
1310     dirsize_status_msg_t dsm;
1311 
1312     memset (&dsm, 0, sizeof (dsm));
1313     status_msg_init (STATUS_MSG (&dsm), _ ("Directory scanning"), 0, dirsize_status_init_cb,
1314                      dirsize_status_update_cb, dirsize_status_deinit_cb);
1315 
1316     for (i = 0; i < panel->dir.len; i++)
1317         if (S_ISDIR (panel->dir.list[i].st.st_mode)
1318             && ((panel->dirs_marked != 0 && panel->dir.list[i].f.marked != 0)
1319                 || panel->dirs_marked == 0)
1320             && !DIR_IS_DOTDOT (panel->dir.list[i].fname->str))
1321         {
1322             vfs_path_t *p;
1323             size_t dir_count = 0;
1324             size_t count = 0;
1325             uintmax_t total = 0;
1326             gboolean ok;
1327 
1328             p = vfs_path_from_str (panel->dir.list[i].fname->str);
1329             ok = compute_dir_size (p, &dsm, &dir_count, &count, &total, FALSE) != FILE_CONT;
1330             vfs_path_free (p, TRUE);
1331             if (ok)
1332                 break;
1333 
1334             panel->dir.list[i].st.st_size = (off_t) total;
1335             panel->dir.list[i].f.dir_size_computed = 1;
1336         }
1337 
1338     status_msg_deinit (STATUS_MSG (&dsm));
1339 
1340     recalculate_panel_summary (panel);
1341 
1342     if (panel->sort_field->sort_routine == (GCompareFunc) sort_size)
1343         panel_re_sort (panel);
1344 
1345     panel->dirty = TRUE;
1346 }

  59 #define check_examine_cd(input, etalon)                                                            \
  60     {                                                                                              \
  61         result = examine_cd (input);                                                               \
  62         ck_assert_msg (strcmp (result->str, etalon) == 0,                                          \
  63                        "\ninput (%s)\nactial (%s) not equal to\netalon (%s)", input, result->str,  \
  64                        etalon);                                                                    \

  66     }

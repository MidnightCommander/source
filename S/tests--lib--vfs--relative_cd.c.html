 126 /* @Test(dataSource = "test_relative_cd_ds") */
 127 START_PARAMETRIZED_TEST (test_relative_cd, test_relative_cd_ds)
     /*   */
 128 {
 129     // given
 130     vfs_path_t *vpath;
 131     int actual_result;
 132 
 133     test_chdir__return_value = 0;
 134 
 135     vpath = vfs_path_from_str_flags (data->input_string, data->input_flags);
 136 
 137     // when
 138     actual_result = mc_chdir (vpath);
 139 
 140     // then
 141     {
 142         const char *element_path;
 143 
 144         ck_assert_int_eq (actual_result, 0);
 145         element_path = vfs_path_get_last_path_str (vpath);
 146         mctest_assert_str_eq (element_path, data->expected_element_path);
 147         vfs_path_free (vpath, TRUE);
 148     }
 149 }
 150 END_PARAMETRIZED_TEST

 156 /* @Test */
 157 START_TEST (test_vpath_to_str_filter)
     /*   */
 158 {
 159     // given
 160     vfs_path_t *vpath, *last_vpath;
 161     char *filtered_path;
 162     const vfs_path_element_t *path_element;
 163 
 164     // when
 165     vpath = vfs_path_from_str ("/test1://some.host/dir");
 166     path_element = vfs_path_element_clone (vfs_path_get_by_index (vpath, -1));
 167     vfs_path_free (vpath, TRUE);
 168 
 169     last_vpath = vfs_path_new (TRUE);
 170 
 171     vfs_path_add_element (last_vpath, path_element);
 172 
 173     filtered_path = vfs_path_to_str_flags (last_vpath, 0,
 174                                            VPF_STRIP_HOME | VPF_STRIP_PASSWORD | VPF_HIDE_CHARSET);
 175 
 176     // then
 177     mctest_assert_str_eq (filtered_path, "test1://some.host/dir");
 178 
 179     vfs_path_free (last_vpath, TRUE);
 180     g_free (filtered_path);
 181 }
 182 END_TEST
